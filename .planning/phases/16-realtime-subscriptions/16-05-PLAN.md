  <name>Task 1: Add coach_id column to task_instances</name>
  <files>supabase/migrations/20260128_add_coach_id_to_task_instances.sql</files>
  <action>
    Create a migration that:
    1. Adds `coach_id UUID REFERENCES auth.users(id)` column to `task_instances`
    2. Populates existing rows by looking up `assignments.assigned_by`
    3. Creates a trigger to auto-populate `coach_id` on INSERT (from assignment)
    4. Adds an index on `coach_id` for efficient filtering
    5. Updates RLS policy for coaches to use the new column directly (simpler, faster)

    The migration should be idempotent (check if column exists before adding).

    SQL structure:
    ```sql
    -- Add coach_id column
    ALTER TABLE task_instances
    ADD COLUMN IF NOT EXISTS coach_id UUID REFERENCES auth.users(id);

    -- Backfill existing data
    UPDATE task_instances ti
    SET coach_id = a.assigned_by
    FROM assignments a
    WHERE ti.assignment_id = a.id
    AND ti.coach_id IS NULL;

    -- Create trigger function for auto-population
    CREATE OR REPLACE FUNCTION set_task_instance_coach_id()
    RETURNS TRIGGER AS $$
    BEGIN
      IF NEW.assignment_id IS NOT NULL THEN
        SELECT assigned_by INTO NEW.coach_id
        FROM assignments WHERE id = NEW.assignment_id;
      END IF;
      RETURN NEW;
    END;
    $$ LANGUAGE plpgsql;

    -- Create trigger
    DROP TRIGGER IF EXISTS set_coach_id_trigger ON task_instances;
    CREATE TRIGGER set_coach_id_trigger
    BEFORE INSERT ON task_instances
    FOR EACH ROW EXECUTE FUNCTION set_task_instance_coach_id();

    -- Add index for realtime filtering
    CREATE INDEX IF NOT EXISTS idx_task_instances_coach_id
    ON task_instances(coach_id);

    -- Update RLS policy to use direct column (faster than subquery)
    DROP POLICY IF EXISTS "Coaches can manage task instances" ON task_instances;
    CREATE POLICY "Coaches can manage task instances"
    ON task_instances FOR ALL
    USING (coach_id = auth.uid());
    ```
  </action>
