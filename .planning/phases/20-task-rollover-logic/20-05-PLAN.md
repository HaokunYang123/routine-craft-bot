Create a hook that shows toast when tasks have been excused:

Per CONTEXT.md:
- Student sees toast: "X tasks were excused by your coach"
- Toast dismisses automatically (1 day expiry)
- Combined if multiple tasks excused

Implementation approach (per RESEARCH.md recommendation):
Since coach and student may be on different devices, we need to check the database for newly excused tasks, not localStorage.

However, for simplicity (MVP):
1. Query for tasks with status = 'excused' that haven't been acknowledged
2. Show toast with count
3. Mark as acknowledged (could add an acknowledged_at field or just track in localStorage which excused task IDs were shown)

Simpler approach:
1. On mount, query for excused tasks for this student
2. Store shown task IDs in localStorage
3. If any new excused tasks (not in localStorage): show toast, update localStorage

```typescript
const SHOWN_EXCUSED_KEY = 'excused-tasks-shown';

export function useExcusedNotification(studentId: string) {
  const { toast } = useToast();
  const hasShownRef = useRef(false);

  useEffect(() => {
    if (!studentId || hasShownRef.current) return;

    const checkExcusedTasks = async () => {
      // Query excused tasks for this student
      const { data: excusedTasks } = await supabase
        .from('task_instances')
        .select('id')
        .eq('assignee_id', studentId)
        .eq('status', 'excused');

      if (!excusedTasks?.length) return;

      // Get previously shown IDs
      const shownIds = JSON.parse(localStorage.getItem(SHOWN_EXCUSED_KEY) || '[]');
      const newExcused = excusedTasks.filter(t => !shownIds.includes(t.id));

      if (newExcused.length > 0) {
        hasShownRef.current = true;
        toast({
          title: 'Tasks Excused',
          description: `${newExcused.length} task${newExcused.length > 1 ? 's were' : ' was'} excused by your coach`,
        });

        // Mark as shown
        const allShown = [...shownIds, ...newExcused.map(t => t.id)];
        localStorage.setItem(SHOWN_EXCUSED_KEY, JSON.stringify(allShown));
      }
    };

    checkExcusedTasks();
  }, [studentId, toast]);
}
```
